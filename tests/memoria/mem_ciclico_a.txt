// Job de memoria A: varredura CICLICA de 7 paginas (Pessoa 2 - Experimento E4)
// Codigo em 3 paginas (0x00, 0x10, 0x20) + dados em 4 paginas (0x01..0x04).
// Cada iteracao referencia as 7 paginas sempre na mesma ordem: padrao
// patologico para FIFO/LRU quando ha menos molduras que paginas.
// Codigo fora da pagina 0 e posicionado via .data (endereco valor).
// Restricao da ISA: BEQ comparando com R0 so salta para 0x00-0x0F.
// ===================================================
100D // 0x00: LOAD R0, [0x0D]   (zero)
110E // 0x01: LOAD R1, [0x0E]   (um)
130F // 0x02: LOAD R3, [0x0F]   (contador = 20)
7305 // 0x03: BEQ  R3, R0, 0x05 (contador == 0 -> HALT)
8100 // 0x04: JUMP 0x100        (corpo do laco, pagina 0x10)
F000 // 0x05: HALT

.data
0D 0000 // zero
0E 0001 // um
0F 0014 // contador = 20 iteracoes
// --- corpo do laco: pagina logica 0x10 ---
100 1210 // 0x100: LOAD R2, [0x10]  (pagina 1)
101 1420 // 0x101: LOAD R4, [0x20]  (pagina 2)
102 8200 // 0x102: JUMP 0x200
// --- continuacao: pagina logica 0x20 ---
200 1530 // 0x200: LOAD R5, [0x30]  (pagina 3)
201 2224 // 0x201: ADD  R2, R2, R4
202 2225 // 0x202: ADD  R2, R2, R5
203 3240 // 0x203: STORE R2, [0x40] (pagina 4, fica suja)
204 4331 // 0x204: SUB  R3, R3, R1
205 8003 // 0x205: JUMP 0x003
// --- dados ---
10 0001
20 0002
30 0003
40 0000 // resultado = 6
